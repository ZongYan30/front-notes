        .get("content", "")
    )
    config["system_prompt"] = system_prompt
    config["name"] = agent_name                 # Nacos 上没有该字段，手动回填
    print(config)
    return AgentSpec.model_validate(config)

## 核心实现
1. 当前我们创建两个agent，一个是web_agent，一个是main_agent。我们可以使用一个抽象类来封装共同的创建逻辑，差异点由其各自的子类来实现。

`src/deal_pilot/agents/agent_factory.py`

```python
from abc import ABC, abstractmethod

import asyncio
from os import name
from typing import Any
from collections.abc import Sequence
from langchain.tools import BaseTool
from langgraph.graph.state import Runnable
from deal_pilot.config.nacos.nacos_agent_spec import AgentSpec, get_agent_spec
from langchain_mcp_adapters.client import MultiServerMCPClient
from deal_pilot.tools import get_tools
from deal_pilot.utils.create_model import create_model
from importlib import import_module

from deal_pilot.utils.func_tools import func_cache

# 定义agent模块路径的映射关系
_AGENT_FACTORIES = {"web_agent": "deal_pilot.agents.web"}


# 定义agent工厂的抽象类
class AgentFactory(ABC):
    # 定义agent工厂的属性
    @property
    @abstractmethod
    def name(self) -> str: ...

    # agent spec配置
    agent_spec_config: AgentSpec | None = None

    # 获取模型
    def _get_model(self) -> Any:
        if self.agent_spec_config is None:
            return create_model()
        return create_model(self.agent_spec_config.model)

    # 获取mcp的tools
    @func_cache(ttl=86400)
    async def _get_mcp_tools(self, mcp_name: str) -> Sequence[BaseTool]:
        if not self.agent_spec_config:
            return []
        if mcp_name not in self.agent_spec_config.mcps:
            return []

        # 请求mcp的tools
        mcp_config = self.agent_spec_config.mcps[mcp_name]
        client = MultiServerMCPClient(
            connections={
                mcp_name: {
                    "transport": "streamable_http",
                    "url": mcp_config.url,
                }
            },
            tool_name_prefix=True,  # 为tools添加前缀，避免冲突
        )
        mcp_tools = await client.get_tools()

        # 处理前缀
        disabled_tool_names = [
            f"{mcp_name}_{name}" for name in mcp_config.disabled_tools
        ]

        tools = [tool for tool in mcp_tools if tool.name not in disabled_tool_names]
        return tools

    # 获取tools和mcp_tools
    async def _get_tools(self) -> Sequence[BaseTool]:
        if not self.agent_spec_config:
            return []
        # 1. 获取本来就有的tools
        tools = get_tools(self.agent_spec_config.tools)
        # 2. 获取mcp_tools
        mcp_tools_groups = await asyncio.gather(
            *[
                self._get_mcp_tools(mcp_name)
                for mcp_name in self.agent_spec_config.mcps.keys()
            ]
        )
        for mcp_tools in mcp_tools_groups:
            tools.extend(mcp_tools)
        return tools

    # 获取子agent
    async def _get_subagents(self) -> list[dict]:
        if not self.agent_spec_config:
            return []

        specs = await asyncio.gather(
            *[
                get_agent_spec(subagent_name)
                for subagent_name in self.agent_spec_config.subagents
            ]
        )

        subagents = []
        for spec in specs:
            if spec.name not in _AGENT_FACTORIES:
                continue
            module = import_module(_AGENT_FACTORIES[spec.name])
            factory = getattr(module, "factory")
            subagents.append(
                {
                    "name": spec.name,
                    "description": spec.description,
                    "runnable": await factory(),
                }
            )

        return subagents

    # 创建agent工厂
    async def factory(self) -> Runnable:
        agent_spec_config = await get_agent_spec(self.name)
        self.agent_spec_config = agent_spec_config

        model = self._get_model()
        tools = await self._get_tools()
        subagents = await self._get_subagents()

        agent = await self.create_agent(
            model=model,
            system_prompt=agent_spec_config.system_prompt,
            tools=tools,
            subagents=subagents,
            skills=agent_spec_config.skills,
        )
        return agent

    # 创建agent，由子类实现差异
    @abstractmethod
    async def create_agent(
        self,
        *,
        model: Any,
        system_prompt: str,
        tools: Sequence[BaseTool],
        subagents: list[dict],
        skills: list[str],
    ) -> Runnable: ...
```

主流程就是 `factory()`：取配置 → `_get_model()` → `_get_tools()` → `_get_subagents()` → `create_agent()`（子类实现）。取子 agent 时用「字符串映射 + `import_module`」动态导入，避免父子模块互相 import 的循环依赖；`_get_mcp_tools` 单独缓存一天（连接开销大），且因为开了 `tool_name_prefix`，过滤 `disabled_tools` 时要拼上同样的 `mcp名_` 前缀。

2. `create_model`修改入参为，和nacos中配置的config一致。

`src/deal_pilot/utils/create_model.py`

```python
from langchain.chat_models import init_chat_model
from deal_pilot.config.nacos.nacos_schemas import AgentModelConfig
from deal_pilot.config.openai_settings import openai_settings
from deal_pilot.utils.shared import get_thread_id


def create_model(model_config: AgentModelConfig | None = None):
    if model_config is None:
        model_config = AgentModelConfig()

    # 初始化模型配置
    config = {
        "model": model_config.name,
        "model_provider": "openai_reasoning",
        "extra_body": {"enable_thinking": model_config.thinking},
    }

    if model_config.temperature is not None:
        config["temperature"] = model_config.temperature

    if model_config.top_p is not None:
        config["top_p"] = model_config.top_p

    # 获取当前线程id
    thread_id = get_thread_id()
    if thread_id:
        config["default_headers"] = {"x-litellm-session-id": thread_id}

    model = init_chat_model(**config)
    return model
```

4. `web_agent`接入抽象类，并且实现自己的创建agent时的差异点。

> web_agent是由langchain的create_agent创建

`src/deal_pilot/agents/web.py`

```python
# 创建web agent工厂
class WebAgentFactory(AgentFactory):
    @property
    def name(self) -> str:
        return "web_agent"

    async def create_agent(
        self,
        *,
        model: Any,
        system_prompt: str,
        tools: Sequence[BaseTool],
        subagents: list[dict],
        skills: list[str],
    ) -> Runnable:
        middlewares = []

        # 添加子agent中间件
        if len(subagents):
            middlewares.append(
                SubAgentMiddleware(
                    backend=StateBackend(),
                    subagents=[CompiledSubAgent(**config) for config in subagents],
                )
            )

        agent = create_agent(
            model=model,
            system_prompt=system_prompt,
            tools=tools,
            middleware=middlewares,
        )
        return agent


web_agent_factory = WebAgentFactory()
factory = web_agent_factory.factory
factory_with_trace = trace_wrapper(factory)
```

差异点：用 langchain 的 `create_agent`，子 agent 走 `SubAgentMiddleware`（需配 `backend=StateBackend()`），且只在有子 agent 时才挂该中间件。

> 两个模块末尾都统一暴露 `factory` / `factory_with_trace`，供 `langgraph.json` 与父级工厂反查使用。
