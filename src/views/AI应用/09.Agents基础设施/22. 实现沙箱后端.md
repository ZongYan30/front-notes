# 实现沙箱后端

## 解决方案

- 自定义实现一个沙箱后端类，继承 deepagents 的 `BaseSandbox` 抽象类
- 重写该抽象类对应的方法，主要是重写异步方法
- 沙箱的后端 id 就使用当前用户的 user_id 即可

`BaseSandbox` 只有 `execute`、`upload_files`、`download_files`、`id` 四个抽象方法，`ls / read / write / edit / delete / grep / glob` 都由它们派生，所以实现这四个就够了。

## 落地方案

1. 实现自定义沙箱类，重写异步的 `执行、上传、下载、删除` 方法。

   ```python [src/deal_pilot/backends/sandbox_backend.py]
   from deepagents.backends.sandbox import BaseSandbox
   from deal_pilot.utils.sandbox import enter_sandbox
   from deal_pilot.utils.shared import get_user_id
   from collections.abc import Callable
   import asyncio
   from deal_pilot.config.nacos.nacos_schemas import OpenSandboxConfig
   from deal_pilot.config.nacos.nacos_config import get_config
   from opensandbox.models.execd import RunCommandOpts
   from datetime import timedelta
   from deepagents.backends.protocol import (
       DeleteResult,
       ExecuteResponse,
       FileDownloadResponse,
       FileUploadResponse,
   )


   type IdGetter = Callable[[], str]


   class SandboxBackend(BaseSandbox):

       id_getter: IdGetter  # 获取当前用户id的函数

       def __init__(self, user_id_getter: IdGetter | None = None):
           super().__init__()
           if not user_id_getter:
               user_id_getter = get_user_id
           self.id_getter = user_id_getter

       # 同步
       @property
       def id(self):
           return self.id_getter()

       def execute(self, command, *, timeout=None):
           return asyncio.run(self.aexecute(command, timeout=timeout))

       def upload_files(self, files):
           return asyncio.run(self.aupload_files(files))

       def download_files(self, paths):
           return asyncio.run(self.adownload_files(paths))

       # 异步
       async def aexecute(self, command, *, timeout=None):
           # 参数归一
           config = await get_config("sandbox_config", OpenSandboxConfig)

           if timeout is None or timeout < 0:
               timeout = config.command_timeout_seconds

           # 获取沙箱
           async with enter_sandbox(self.id) as sandbox:
               execution = await sandbox.commands.run(
                   command, opts=RunCommandOpts(timeout=timedelta(seconds=timeout))
               )

           return ExecuteResponse(
               output=str(execution), exit_code=execution.exit_code, truncated=False
           )

       async def aupload_files(
           self, files: list[tuple[str, bytes]]
       ) -> list[FileUploadResponse]:
           responses: list[FileUploadResponse] = []

           async with enter_sandbox(self.id) as sandbox:
               for path, data in files:
                   try:
                       await sandbox.files.write_file(path=path, data=data)
                       responses.append(FileUploadResponse(path=path))
                   except Exception as e:
                       responses.append(FileUploadResponse(path=path, error=str(e)))

           return responses

       async def adownload_files(self, paths: list[str]) -> list[FileDownloadResponse]:
           responses: list[FileDownloadResponse] = []

           async with enter_sandbox(self.id) as sandbox:
               for path in paths:
                   try:
                       content = await sandbox.files.read_bytes(path)
                       responses.append(FileDownloadResponse(path=path, content=content))
                   except Exception as e:
                       responses.append(FileDownloadResponse(path=path, error=str(e)))

           return responses

       async def adelete(self, file_path: str) -> DeleteResult:
           res = await self.aexecute(f"rm -rf -- {file_path}")
           if res.exit_code == 0:
               return DeleteResult(path=file_path)
           return DeleteResult(
               error=f"删除路径{file_path}失败，原因是:{res.output}\n退出码：{res.exit_code}"
           )
   ```

   > 同步方法只是 `asyncio.run` 转调异步版，真正逻辑都在 `a*` 方法里。

2. 封装`获取用户:get_user_id`

   ```python [src/deal_pilot/utils/shared.py]
   from langgraph.config import get_config

   # 获取当前线程id
   def get_thread_id():
       config = get_config()
       if not config:
           return None
       return config.get("configurable", {}).get("thread_id", None)


   # 获取当前用户id
   def get_user_id():
       config = get_config()
       user_id=config.get("configurable", {}).get("langgraph_auth_user_id",None)
       if not user_id:
           raise RuntimeError("无法通过langgraph_auth_user_id获取当前认证用户的ID")
       return user_id
   ```

   > 用户身份取自 langgraph `configurable.langgraph_auth_user_id`，取不到直接抛错，避免所有用户共用同一个沙箱。

3. 封装沙箱异步上下文管理器`enter_sandbox`，先延时，再执行，最后再延长存活和关闭沙箱连接(多处用到)

   ```python [src/deal_pilot/utils/sandbox.py]
   # 沙箱异步上下文管理器
   @asynccontextmanager
   async def enter_sandbox(key: str | None = None, store: BaseStore | None = None):
       sandbox= await get_sandbox(key, store)
       config = await get_config("sandbox_config", OpenSandboxConfig)
       await sandbox.renew(timeout=timedelta(seconds=config.command_timeout_seconds))
       yield sandbox
       await sandbox.renew(timeout=timedelta(seconds=config.command_timeout_seconds))
       await sandbox.close()
   ```

   > 进入 / 退出各 `renew` 一次（干活期间不过期、用完再活一会儿方便下次复用），最后 `close` 只断连接不销毁容器，数据留在卷柜里。

## 测试沙箱后端

```python
# 创建一个沙箱后端
from deal_pilot.backends.sandbox_backend import SandboxBackend

sandbox = SandboxBackend(lambda:"abc")
sandbox.id
```

```python
# 测试往沙箱上传文件
await sandbox.aupload_files([
    ("/root/workspace/1.txt", b"Hello, 1.txt"),
    ("/root/workspace/2.txt", b"Hello, 2.txt"),
    ("/root/workspace/nested/3.txt", b"Hello, 3.txt"),
])
```

```python
# 测试从沙箱下载文件
await sandbox.adownload_files([
    "/root/workspace/1.txt",
    "/root/workspace/2.txt",
    "/root/workspace/nested/3.txt"
])
```

```python
await sandbox.aexecute("ls ../ -l")
```

```python
await sandbox.adelete("/root/workspace/nested")
```

> `SandboxBackend(lambda:"abc")` 注入了固定 key（不取真实 user_id），多次运行都命中同一个沙箱，方便验证复用。
