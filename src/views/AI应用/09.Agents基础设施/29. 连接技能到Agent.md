# 连接技能到Agent
沙箱中的技能都准备好了，接下来就是将技能挂载到agent中。

## 解决方案
1. 所有的agent都使用`deepagents`的`create_deep_agent`函数创建
2. agent工厂函数负责统一的创建，但是可以将一些不同点下方到子类中实现(比如:schema、subagent、middleware等)
3. main_agent和web_agent的重新接入agent工厂函数
4. nacos配置中心新增特殊目录提示词，并且完善对应的agent_spec配置。


## 落地方案

2. agent工厂函数负责统一创建:
```python
from abc import ABC, abstractmethod

import asyncio
from os import name
from typing import Any
from collections.abc import Sequence
from langchain.tools import BaseTool
from langgraph.graph.state import Runnable
from deal_pilot.config.nacos.nacos_agent_spec import AgentSpec, get_agent_spec
from langchain_mcp_adapters.client import MultiServerMCPClient
from deal_pilot.tools import get_tools
from deal_pilot.utils.create_model import create_model
from importlib import import_module
from deepagents import (
    AsyncSubAgent,
    CompiledSubAgent,
    DeepAgentState,
    SubAgent,
    create_deep_agent,
)
from langchain.agents.middleware import AgentMiddleware
from deal_pilot.utils.func_tools import func_cache
from deepagents.backends import BackendProtocol
from deal_pilot.backends.sandbox_backend import SandboxBackend

# 定义agent模块路径的映射关系
_AGENT_FACTORIES = {"web_agent": "deal_pilot.agents.web"}


# 定义agent工厂的抽象类
class AgentFactory(ABC):
    # 定义agent工厂的属性
    @property
    @abstractmethod
    def name(self) -> str: ...

    # agent spec配置
    agent_spec_config: AgentSpec | None = None

    # 获取模型
    def _get_model(self) -> Any:
        if self.agent_spec_config is None:
            return create_model()
        return create_model(self.agent_spec_config.model)

    # 获取mcp的tools
    @func_cache(ttl=86400)
    async def _get_mcp_tools(self, mcp_name: str) -> Sequence[BaseTool]:
        if not self.agent_spec_config:
            return []
        if mcp_name not in self.agent_spec_config.mcps:
            return []

        # 请求mcp的tools
        mcp_config = self.agent_spec_config.mcps[mcp_name]
        client = MultiServerMCPClient(
            connections={
                mcp_name: {
                    "transport": "streamable_http",
                    "url": mcp_config.url,
                }
            },
            tool_name_prefix=True,  # 为tools添加前缀，避免冲突
        )
        mcp_tools = await client.get_tools()

        # 处理前缀
        disabled_tool_names = [
            f"{mcp_name}_{name}" for name in mcp_config.disabled_tools
        ]

        tools = [tool for tool in mcp_tools if tool.name not in disabled_tool_names]
        return tools

    # 获取tools和mcp_tools
    async def _get_tools(self) -> Sequence[BaseTool]:
        if not self.agent_spec_config:
            return []
        # 1. 获取本来就有的tools
        tools = get_tools(self.agent_spec_config.tools)
        # 2. 获取mcp_tools
        mcp_tools_groups = await asyncio.gather(
            *[
                self._get_mcp_tools(mcp_name)
                for mcp_name in self.agent_spec_config.mcps.keys()
            ]
        )
        for mcp_tools in mcp_tools_groups:
            tools.extend(mcp_tools)
        return tools

    # 获取子agent
    async def _get_subagents(self) -> list[dict]:
        if not self.agent_spec_config:
            return []

        specs = await asyncio.gather(
            *[
                get_agent_spec(subagent_name)
                for subagent_name in self.agent_spec_config.subagents
            ]
        )

        subagents = []
        for spec in specs:
            if spec.name not in _AGENT_FACTORIES:
                continue
            module = import_module(_AGENT_FACTORIES[spec.name])
            factory = getattr(module, "factory")
            subagents.append(
                {
                    "name": spec.name,
                    "description": spec.description,
                    "runnable": await factory(),
                }
            )

        return subagents

    # 创建agent工厂
    async def factory(self) -> Runnable:
        agent_spec_config = await get_agent_spec(self.name)
        self.agent_spec_config = agent_spec_config

        model = self._get_model()
        tools = await self._get_tools()
        subagents = await self._get_subagents()

        agent = await self.create_agent(
            model=model,
            system_prompt=agent_spec_config.system_prompt,
            tools=tools,
            subagents=subagents,
            skills=agent_spec_config.skills,
        )
        return agent

    # 创建agent，子类可以重写
    async def create_agent(
        self,
        *,
        model: Any,
        system_prompt: str,
        tools: Sequence[BaseTool],
        subagents: list[dict],
        skills: list[str],
    ) -> Runnable:
        config = {
            "model": model,
            "system_prompt": system_prompt,
            "tools": tools,
            "state_schema": self.get_state_schema(),
            "subagents": self.get_subagents(subagents),
            "middleware": self.get_middlewares(),
            "backend": self.get_backend(),
        }

        if len(skills):
            config["skills"] = [f"/shared/skills/{self.name}"]

        agent = create_deep_agent(**config)
        return agent

    # 获取schema，子类可以重写
    def get_state_schema(self) -> type[DeepAgentState] | None:
        return DeepAgentState

    # 获取子agent，子类可以重写
    def get_subagents(
        self, subagent_configs: list[dict]
    ) -> Sequence[SubAgent | CompiledSubAgent | AsyncSubAgent] | None:
        return [CompiledSubAgent(**config) for config in subagent_configs]

    # 获取中间件，子类可以重写
    def get_middlewares(self) -> Sequence[AgentMiddleware]:
        return []

    # 获取后端，子类可以重写
    def get_backend(self) -> BackendProtocol | None:
        return SandboxBackend()
```

4. nacos配置中心新增提示词:
`special_dir`
```text
## 特殊目录说明

### /shared

位于 `/shared` 的目录是共享目录，每个用户级沙箱中的 `/shared` 目录完全一致。

为了保证用户不互相干扰，因此，该目录被设置为 **只读**，你无法在修改、新增、删除该目录中的任何内容。

虽然不会限制你执行该目录中的代码，但值得注意的是，如果执行代码会对该目录造成影响，比如：

- 执行代码需要先在代码位置安装依赖
- 执行代码会在目录中新增文件
- ...

在这些情况下，执行代码均会失败。

所以，如果你需要在该目录中执行代码，强烈建议将该目录中的代码文件，或代码所在的目录，拷贝到一个临时目录去执行。
```

## 修复bugs
之前的沙箱创建完之后忘记记录缓存了，会导致启动`make dev`之后，创建多个沙箱。
正常期望只有两个："专有沙箱"和"共享沙箱"。

sandbox中创建时传入key和store，创建完毕之后记录sandbox_id到store中。
utils/sandbox.py
```python
from datetime import timedelta
import uuid
from langgraph.store.base import BaseStore
from langgraph.store.memory import InMemoryStore
from langgraph.runtime import get_runtime
import asyncio
from weakref import WeakValueDictionary
from opensandbox import Sandbox
from opensandbox.config import ConnectionConfig
from deal_pilot.config.nacos.nacos_config import get_config
from deal_pilot.config.nacos.nacos_schemas import OpenSandboxConfig
from deal_pilot.config.sandbox_settings import sandbox_settings
from opensandbox.models import PVC, Volume
from opensandbox.exceptions import SandboxApiException
from contextlib import asynccontextmanager
from pydantic import BaseModel, Field

_lockMap: WeakValueDictionary[str, asyncio.Lock] = WeakValueDictionary()  # 并发锁
_MEMORY_STORE = InMemoryStore()  # 存储store
_NAMESPACE = ("__sandbox__",)  # store存储命名空间
SHARED_VOLUME_NAME = "sandbox-shared-data"  # 沙箱共享卷名
SHARED_SANDBOX_KEY = "sandbox-shared"  # 共享沙箱key


# 卷柜配置
class VolumeConfig(BaseModel):
    mount_path: str
    name: str
    readonly: bool = Field(default=False)


# 获取卷柜名称（从store中获取）
async def _get_store(key: str, store: BaseStore, field_name: str) -> str | None:
    dict = await store.aget(_NAMESPACE, key)
    if dict is None:
        return None
    return dict.value.get(field_name)


# 创建沙箱
async def _create(
    *,
    key: str,
    store: BaseStore,
    sandbox_config: OpenSandboxConfig,
    connection_config: ConnectionConfig,
    volume_configs: list[VolumeConfig] | None = None,
):

    # 创建卷柜
    if volume_configs is None:
        volume_configs = []

    volumes = [
        Volume(
            name=v.name,
            mountPath=v.mount_path,
            readOnly=v.readonly,
            pvc=PVC(
                claimName=v.name,
                createIfNotExists=True,
                deleteOnSandboxTermination=False,
            ),
        )
        for v in volume_configs
    ]

    # 创建沙箱
    sandbox = await Sandbox.create(
        # 镜像
        image=sandbox_config.image,
        # 连接配置
        connection_config=connection_config,
        # 超时时间
        timeout=timedelta(seconds=sandbox_config.timeout_seconds),
        # 就绪超时时间
        ready_timeout=timedelta(seconds=sandbox_config.ready_timeout_seconds),
        # 卷柜
        volumes=volumes,
    )
    # 把 sandbox_id 写回 store，供后续 _connect 复用
    await store.aput(_NAMESPACE, key, {"sandbox_id": sandbox.id})
    return sandbox


# 连接沙箱
async def _connect(
    *,
    connection_config: ConnectionConfig,
    key: str,
    store: BaseStore,
) -> Sandbox | None:
    # 从store中取出sandbox_id
    sandbox_id = await _get_store(key, store, "sandbox_id")
    if sandbox_id is None:
        return None

    # 连接沙箱（沙箱有可能已经过期了，连接不上就会报错）
    try:
        sandbox = await Sandbox.connect(
            sandbox_id=sandbox_id, connection_config=connection_config
        )
        return sandbox
    except SandboxApiException as exc:
        if exc.status_code == 404:
            return None
        else:
            raise


# 获取沙箱
async def get_sandbox(
    key: str | None = None,
    store: BaseStore | None = None,
    volume_configs: list[VolumeConfig] | None = None,
):
    # 参数归一化
    if key is None:
        key = str(uuid.uuid4())
    if store is None:
        try:
            runtime = get_runtime()
            store = runtime.store
            if store is None:
                store = _MEMORY_STORE
        except Exception:
            store = _MEMORY_STORE

    # 加锁控制并发,有就取出来,没有就默认创建一个锁
    lock = _lockMap.setdefault(key, asyncio.Lock())

    async with lock:
        # print(key, flush=True)
        # await asyncio.sleep(5)
        sandbox_config = await get_config("sandbox_config", OpenSandboxConfig)
        connection_config = ConnectionConfig(
            domain=sandbox_settings.domain,
            api_key=sandbox_settings.api_key,
            use_server_proxy=True,
        )

        sandbox = await _connect(
            connection_config=connection_config,
            key=key,
            store=store,
        )

        if not sandbox:
            sandbox = await _create(
                sandbox_config=sandbox_config,
                connection_config=connection_config,
                volume_configs=volume_configs,
                store=store,
                key=key,
            )

        return sandbox


# 获取共享沙箱
async def get_shared_sandbox(store: BaseStore | None = None) -> Sandbox:
    return await get_sandbox(
        SHARED_SANDBOX_KEY,
        store,
        [
            VolumeConfig(
                name=SHARED_VOLUME_NAME,
                mount_path="/shared",
                readonly=False,
            )
        ],
    )


# 沙箱异步上下文管理器
@asynccontextmanager
async def enter_sandbox(
    key: str | None = None,
    store: BaseStore | None = None,
    volume_configs: list[VolumeConfig] | None = None,
):
    sandbox = await get_sandbox(key, store, volume_configs)
    config = await get_config("sandbox_config", OpenSandboxConfig)
    await sandbox.renew(timeout=timedelta(seconds=config.command_timeout_seconds))
    yield sandbox
    await sandbox.renew(timeout=timedelta(seconds=config.command_timeout_seconds))
    await sandbox.close()
```
