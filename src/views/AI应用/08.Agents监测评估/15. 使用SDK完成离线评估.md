## 离线评估

通常，使用代码的方式创建评估器会具有更大的灵活性。

1. 认识一下langfuse的`run_experiment`:

```python
from langfuse import get_client

langfuse = get_client()

langfuse.run_experiment(
    # 给实验取个名字
    name=name,
    # 如果不配置，则使用name+当前时间自动生成
    run_name=None,
    description=description,
    # 此次实验针对哪些数据项
    data=items,
    # 评估器列表 (接收函数)
    evaluators=[content_similarity, emotional_similarity],
    # 综合评估器 (接收函数)
    composite_evaluator=weighted_average_evaluator,
    task=experiment_task,  # type: ignore
)
```

evaluators: 评估器列表 (接收返回值为Evaluation对象的函数)
composite_evaluator: 综合评估器 (接收返回值为Evaluation对象的函数)，往往用来综合多个评估器的结果。比如求权重。

2. `Evaluation`函数可以创建一个评估器，类似于手动操作`langfuse`控制面的新增评估器。

```python
from langfuse import Evaluation

# 定义一个评估器
def evaluatorFn( *, input: str, output: str, expected_output: str, **kwargs):
    """
    参数在执行experiment时会自动传递给评估器。
    参数:
    - input: 用户输入
    - output: 实际输出
    - expected_output: 期望输出
    - kwargs: 其他参数
    """
    # 其他代码(调用agent来进行评估，提供提示词，)
    return Evaluation(
        name="相似度评估器",
        value=0,# 该次评估器的分数
        comment="评估器的内容输出",
        data_type="NUMERIC",
    )
```

## 示例-创建相似度评估器

接下来示例一个相似度评估器,目标如下：

1. 评估器调用模型，模型需要返回`response_format`的格式

```python
from pydantic import BaseModel, Field


class EvaluationResult(BaseModel):
    score: float = Field(ge=0, le=1, description="当前维度的评分")
    reason: str = Field(min_length=1, description="用一句话说明评分的主要原因")
```

2. 封装一个创建评估器的函数

```python
from langfuse import Evaluation
from ai_eval.utils.create_model import create_model
from ai_eval.benchmarks.evaluators.schemas import EvaluationResult
from langchain.agents import create_agent
from langchain.agents.structured_output import ToolStrategy
from langchain.messages import HumanMessage

# 创建评估器
def create_score_evaluator(prompt_template, evaluator_name):

    async def content_similarity(
        *, input: str, output: str, expected_output: str, **kwargs
    ):
        if not output:
            return Evaluation(name=evaluator_name,value=0,comment="没有实际输出",data_type="NUMERIC",)

        model = create_model()
        agent = create_agent(
            model=model, response_format=ToolStrategy(EvaluationResult)
        )

        # 填充提示词的模板字符串
        prompt = prompt_template.format(
            input=input, output=output, expected_output=expected_output
        )

        result = await agent.ainvoke(input={"messages": [HumanMessage(prompt)]})

        # 从response_format中提取score和reason
        score = result["structured_response"].score
        reason = result["structured_response"].reason
        return Evaluation(
            name=evaluator_name, value=score, comment=reason, data_type="NUMERIC"
        )

    return content_similarity

```

3. 相似度评估器

```python
from ai_eval.benchmarks.evaluators.create_score_evaluator import create_score_evaluator

_NAME = "内容相似度"

_PROMPT = """你是一名客观的答案评估员。只评价实际输出对期望输出中关键内容的覆盖程度，给出 0 到 1 的分数。

评分原则：
1. 从期望输出中找出关键事实、步骤和要求，判断实际输出正确覆盖了多少。
2. 语义等价即可，不要求措辞、语序或格式相同。
3. 遗漏越多，分数越低；说反或说错的要点不能算作已覆盖。
4. 不因合理的补充内容扣分；本维度只看期望要点是否得到正确覆盖。
5. 只依据提供的内容评分，不自行补充事实。

分数参考：

- 1.0：所有关键要点均已覆盖。
- 0.75：仅有轻微遗漏。
- 0.5：约有一半关键内容被覆盖。
- 0.25：仅覆盖少量要点。
- 0.0：没有覆盖关键内容或实际输出为空。

请用一句话说明最重要的覆盖或遗漏情况。

用户输入：
{input}

期望输出：
{expected_output}

实际输出：
{output}"""

content_similarity = create_score_evaluator(_PROMPT, _NAME)
```

from typing import List

from langfuse import get_client
from langfuse.api import Dataset

langfuse = get_client()


def get_all_datasets():
    page = 1
    datasets: List[Dataset] = []
    while True:
        result = langfuse.api.datasets.list(page=page, limit=100)
        datasets.extend(result.data)
        if page >= result.meta.total_pages:
            break
        page += 1

    return datasets


def get_items(dataset_name: str):
    dataset = langfuse.get_dataset(dataset_name)
    return dataset.items

from langfuse import Evaluation
from ai_eval.utils.create_model import create_model
from ai_eval.benchmarks.evaluators.schemas import EvaluationResult
from langchain.agents import create_agent
from langchain.agents.structured_output import ToolStrategy
from langchain.messages import HumanMessage


def create_score_evaluator(prompt_template, evaluator_name):

    async def content_similarity(
        *, input: str, output: str, expected_output: str, **kwargs
    ):
        if not output:
            return Evaluation(
                name=evaluator_name,
                value=0,
                comment="没有实际输出",
                data_type="NUMERIC",
            )
        model = create_model()
        agent = create_agent(
            model=model, response_format=ToolStrategy(EvaluationResult)
        )

        prompt = prompt_template.format(
            input=input, output=output, expected_output=expected_output
        )
        result = await agent.ainvoke(input={"messages": [HumanMessage(prompt)]})
        score = result["structured_response"].score
        reason = result["structured_response"].reason
        return Evaluation(
            name=evaluator_name, value=score, comment=reason, data_type="NUMERIC"
        )

    return content_similarity

from ai_eval.benchmarks.evaluators.create_score_evaluator import create_score_evaluator

_NAME = "情绪相似度"

_PROMPT = """你是一名客观的情绪价值相似度评估员。结合用户输入，比较实际输出与期望输出带给用户的情绪回应，给出 0 到 1 的分数。

评分原则：
1. 关注是否识别并接住用户当下的情绪，以及回应的态度、支持方式和情绪强度是否接近期望输出。
2. 语义和效果相近即可，不要求复用期望输出的措辞、表情或句式；不要因为表情更多、赞美更热烈就给高分。
3. 用户需要安慰、共鸣、庆祝或明确立场时，判断实际输出是否给出了相应的回应；用户需要认真提醒或直接答案时，也要尊重期望输出的克制程度。
4. 机械复述、说教、空泛安慰、敷衍、无端指责或虚构亲密经历，会降低相似度；若期望输出包含必要的风险关切，实际输出忽略或淡化它应给低分。
5. 只依据用户输入及提供的两段输出评分，不自行补充事实；不要把内容要点覆盖率当作本维度的分数。

分数参考：

- 1.0：情绪判断、回应态度和支持方式与期望输出高度一致。
- 0.75：整体情绪效果接近，仅有轻微语气或力度偏差。
- 0.5：回应了部分情绪需求，但支持方式或力度有明显偏差。
- 0.25：只有少量情绪回应，主要态度与期望输出不符。
- 0.0：情绪回应完全不符、忽略关键风险关切，或实际输出为空。

请用一句话说明最主要的情绪回应差异。

用户输入：
{input}

期望输出：
{expected_output}

实际输出：
{output}"""

```python [src/ai_eval/benchmarks/evaluators/weighted_average.py]
from langfuse import Evaluation

SCORE_WEIGHT = {"内容相似度": 0.4, "情绪相似度": 0.6}


def weighted_average_evaluator(
    *, evaluations: list[Evaluation], **kwargs
) -> list[Evaluation]:
    scores = {eval.name: float(eval.value) for eval in evaluations}
    score = sum(scores[name] * weight for name, weight in SCORE_WEIGHT.items())

    return [
        Evaluation(
            name="总体相似度",
            value=score,
            comment=",".join(
                [f"{name}权重为{weight*100}%" for name, weight in SCORE_WEIGHT.items()]
            ),
            data_type="NUMERIC",
        )
    ]
```

```python [src/ai_eval/benchmarks/experiment.py]
from langfuse.api import DatasetItem
from langfuse import get_client
from ai_eval.benchmarks.dataset import get_items
from langgraph_sdk import get_client as get_agent_client
from ai_eval.benchmarks.evaluators.weighted_average import weighted_average_evaluator
from ai_eval.config.benchmark_settings import benchmark_settings
from opentelemetry import trace
from ai_eval.benchmarks.evaluators.content_similarity import content_similarity
from ai_eval.benchmarks.evaluators.emotional_similarity import emotional_similarity

langfuse = get_client()


async def experiment_task(*, item: DatasetItem, **kwargs):
    trace_id = langfuse.get_current_trace_id()  # 获取traceid
    parent_span_id = langfuse.get_current_observation_id()  # 获取当前的parent_span_id
    span_attributes = getattr(trace.get_current_span(), "attributes", {})
    env = span_attributes.get("langfuse.environment")
    async with get_agent_client(
        url=benchmark_settings.agent_server,
        headers={"Authorization": f"Bearer {benchmark_settings.user_id}"},
    ) as client:
        result = await client.runs.wait(
            thread_id=None,
            assistant_id=benchmark_settings.graph_id,
            input={"messages": [{"role": "user", "content": item.input}]},
            metadata={
                "trace_id": trace_id,
                "parent_span_id": parent_span_id,
                "trace_environment": env,
            },
        )
    last_message = result.get("messages")[-1]  # type: ignore
    return last_message["content"]


def run_experiment(name: str, dataset_name: str, description: str | None = ""):
    items = get_items(dataset_name)
    # 运行实验
    langfuse.run_experiment(
        # 给实验取个名字
        name=name,
        # 如果不配置，则使用name+当前时间自动生成
        run_name=None,
        description=description,
        # 此次实验针对哪些数据项
        data=items,
        evaluators=[content_similarity, emotional_similarity],
        composite_evaluator=weighted_average_evaluator,
        task=experiment_task,  # type: ignore
    )
```
