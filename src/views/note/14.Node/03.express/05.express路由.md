# express路由
在真实的开发下，我们并不会将一个又一个的请求给散落，这样不易于维护。
来看下标准的路由结构目录。

```text
routes/
├─ init.js            # 路由初始化
├─ api/
│  ├─ students.js     # 学生路由
│  └─ users.js        # 用户路由
├─ utils/
└─ middleware/
   └─ error.js        # 错误处理中间件
```

## 使用方法
```js
const express = require("express");
const router = express.Router();//创建一个路由实例

router.get("/add", (req, res) => {
   //baseUrl是当前路由实例的根路径，path是请求的路径
   console.log(req.baseUrl);// /api/v1/student
   console.log(req.path);// /add  (实际上/api/v1/student/add)
   
 })

router.post("/", (req, res) => { })

app.use("/api/v1/student", router);//这里可以看出,路由实例本质就是一个中间件
```

## 案例-学生路由
在一个标准的工程下，具有
* 统一的返回后端的消息格式
* 统一错误处理中间件
* 路由初始化
* 对应的业务路由

:::code-group
```js [utils.js]
/**
 * 统一错误处理
 * @param {*} err 
 * @param {*} errCode 
 * @returns 
 */
exports.getErr = function (err = "server internal error", errCode = 500) {
    return {
        code: errCode,
        msg: err,
    };
};

/**
 * 统一响应处理
 */
exports.getResult = function (result) {
    return {
        code: 0,
        msg: "",
        data: result,
    };
};

/**
 * 异步处理函数
 * @param {*} handler 
 * @returns 
 */
exports.asyncHandler = (handler) => {
    return async (req, res, next) => {
        try {
            const result = await handler(req, res, next);
            res.send(exports.getResult(result));
        } catch (err) {
            next(err);
        }
    };
};
```

```js [middleware/error.js]
const utils = require('../utils');

module.exports = (err, req, res, next) => {
    if (err) {
        const errObj = err instanceof Error ? err.message : err;
        res.status(500).send(utils.getErr(errObj));
    } else {
        next();
    }
}
```

```js [api/students.js]
/* 学生相关的路由 */
const express = require("express");
const { asyncHandler } = require('../utils');
const studentService = require("../../services/studentService");

const router = express.Router();

// 获取所有学生
router.get("/", asyncHandler(async (req, res) => {
    const { page, size, name, sex } = req.query;
    console.log(page, size, name, sex);
    return await studentService.getStudents(page, size, sex, name);
}))

// 通过id获取学生
router.get("/:id", asyncHandler(async (req, res) => {
    console.log(req.params);
    return await studentService.getStudentById(req.params.id);
}))

// 添加学生
router.post("/", asyncHandler(async (req, res) => {
    console.log(req.body);
    return await studentService.addStudent(req.body);
}))

// 修改学生
router.put("/:id", asyncHandler(async (req, res) => {
    console.log(req.params, req.body);
    return await studentService.updateStudent(req.params.id, req.body);
}))

// 删除学生
router.delete("/:id", asyncHandler(async (req, res) => {
    console.log(req.params);
    return await studentService.delStudent(req.params.id);
}))

module.exports = router;
```

```js [routes/init.js]
/* 路由初始化 */
const path = require('path');
const express = require('express');

const app = express();
const staticPath = path.resolve(__dirname, "../public");

app.use("/static", express.static(staticPath));
app.use(express.urlencoded({ extended: true }));
app.use(express.json());

// 路由
app.use("/api/v1/student", require("./api/students"));

app.use(require("./middleware/error"));

app.listen(3000, () => {
    console.log('server is running at port 3000');
})
```
