# 文件上传

## 文件上传原理
### 1. 文件上传格式
 文件上传的mime格式一般是`multipart/form-data`
 此格式特点:
 * 具有分隔符`--`
 * 每个字段之间用空行隔开
 ```sh
--boundary123
Content-Disposition: form-data; name="file"; filename="example.jpg"
Content-Type: image/jpeg

(二进制文件数据...)
--boundary123
Content-Disposition: form-data; name="description"
 ```

## 服务端
服务端接收文件上传数据，需要使用第三方库 [multer](https://github.com/expressjs/multer#readme)

2. 上传文件功能 `upload.js`
```js
const express = require("express");
const multer = require("multer");
const path = require("path");

const router = express.Router();

//配置文件存储路径
const storage = multer.diskStorage({
  //存储路径
  destination: function (req, file, cb) {
    cb(null, path.resolve(__dirname, "../../public/upload"));
  },
  // 配置上传的文件名
  filename: function (req, file, cb) {
    // 时间戳-6位随机字符.文件后缀
    const timeStamp = Date.now();
    const ramdomStr = Math.random().toString(36).slice(-6);
    const ext = path.extname(file.originalname);
    const filename = `${timeStamp}-${ramdomStr}${ext}`;
    cb(null, filename);
  },
});


const upload = multer({
  storage,
  limits: { fileSize: 150 * 1024},//文件大小限制
  // 验证文件格式  
  fileFilter(req, file, cb) {
    //验证文件后缀名
    const extname = path.extname(file.originalname);
    const whitelist = [".jpg", ".gif", "png"];
    if (whitelist.includes(extname)) {
      cb(null, true);
    } else {
      cb(new Error(`your ext name of ${extname} is not support`));
    }
  },
});

// 路由
router.post("/", upload.single("img"), (req, res) => {
  const url = `/upload/${req.file.filename}`;
  res.send({
    code: 0,
    msg: "",
    data: url,
  });
});

module.exports = router;
```

3. 使用路由
```js
app.use("/api/v1/upload", require("./api/upload"));
```

4. 错误中间件--处理上传文件的错误
```js
// 处理错误的中间件
const getMsg = require("./getSendResult");
const multer = require("multer");

module.exports = (err, req, res, next) => {
  if (err) {
    // 处理multer错误
    if (err instanceof multer.MulterError) {
      res.status(200).send(getMsg.getErr(err.message));
      return;
    }

    //处理其他错误
    const errObj = err instanceof Error ? err.message : err;
    res.status(500).send(getMsg.getErr(errObj));
  } else {
    next();
  }
};
```

## 客户端
客户端上传文件有两种方法,`form`表单或者`formData`API
1. form表单的格式
```html
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Document</title>
  </head>
  <body>
    <!--
     action：上传api地址 
     enctype：上传文件格式
     -->
    <form action="/api/upload" method="POST" enctype="multipart/form-data">
      <p>
        <input type="text" name="a" />
      </p>
      <p>
        <input type="file" name="img" />
      </p>
      <p>
        <button>提交</button>
      </p>
    </form>
  </body>
</html>
```

2. formData的格式
```html
<!DOCTYPE html>
<html lang="en">

<head>
  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0" />
  <title>Document</title>
</head>

<body>
  <p>
    <input type="text" name="a" />
  </p>
  <p>
    <input type="file" name="img" accept="image/*" multiple />
  </p>
  <p>
    <button>提交</button>
  </p>
  <img src="" alt="" />
  <script>
    // 上传函数
    function upload() {
      const inpA = document.querySelector("[name=a]");
      const inpFile = document.querySelector("[name=img]");
      const img = document.querySelector("img");

      //借助formData构建请求body
      const formData = new FormData();

      formData.append("a", inpA.value);

      for (const file of inpFile.files) {
        formData.append("img", file, file.name);
      }

      fetch("/api/upload", {
        body: formData,
        method: "POST",
      }).then((resp) => resp.json())
      .then((resp) => {
          console.log(resp);
         // 提示给前端  
          if (resp.code) {
            alert(resp.msg);
            return
          } 
          //展示到页面   
          img.src = resp.data;
        });
    }

    document.querySelector("button").onclick = upload;
  </script>
</body>

</html>
```

## 注意
* 一般上传文件在服务端会添加`gitignore`文件,来忽略上传的文件
```
node_modules/
temp/
~*.ppt*
logs/
.DS_Store
public/upload
```

* 服务端必须做验证，其余前端做的上传文件的验证都只是友好行为。
