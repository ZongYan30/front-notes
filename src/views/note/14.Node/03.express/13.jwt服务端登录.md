# jwt实现登录服务端
## 安装jsonwebtoken

[jsonwebtoken](https://github.com/auth0/node-jsonwebtoken#readme)

```bash
npm install jsonwebtoken -S
```

## 使用方法
```js
const jwt = require('jsonwebtoken');

jwt.sign(info, secret, { expiresIn: maxAge });//生成token

jwt.verify(token, secret);//验证token
```


## 实现jwt
创建`jwt.js`文件，封装两个函数，生成token和验证token：

```js
const jwt = require('jsonwebtoken');
const secret = "zy-node-demo";// 密钥
const cookieKey = "token";//cookie名

/* 发布生成token */
exports.publish = (res, maxAge = 3600 * 24, info) => {
    const token = jwt.sign(info, secret, { expiresIn: maxAge });//生成token
    res.header("authorization", `${token}`);
    res.cookie(cookieKey, token, {
        path: "/",
        maxAge: maxAge * 1000
    })
    console.log("生成的jwt==>", token);
}

/* 验证token */
exports.verify = (req) => {
    let token = req.headers.authorization || req.cookies[cookieKey];

    if (!token) { return null };

    // 截取token Barerer yourToken
    const tokenStr = token.split(" ");
    token = tokenStr.length == 1 ? tokenStr[0] : tokenStr[1];

    try {
        const decoded = jwt.verify(token, secret);
        return decoded;
    } catch (error) {
        console.log("token验证失败", error);
        return null;
    }
}
```

登录的时候创建token，发送给客户端

`services/adminService.js`

```js
const express = require('express')
const { getErr, getResult } = require('../utils');
const jwtUtils = require('../../utils/jwt');
const adminService = require('../../services/adminService');
const router = express.Router();

router.post("/login", async (req, res) => {
    const { loginId, password } = req.body;
    const result = await adminService.login(loginId, password);
    console.log("登录==>", result);

    if (!result) {
        res.status(401).send(getErr("登录失败", 401));
        return
    }

    jwtUtils.publish(res, 3600, { userId: result.id });//生成jwt相关操作
    res.send(getResult(result, `登录成功${loginId}`));
})


module.exports = router;
```

中间件校验jwt

`routes/middleware/authToken.js`

```js
/* 验证token中间件 */
const { getErr } = require('../utils');
const { pathToRegexp } = require("path-to-regexp");
const jwtUtils = require('../../utils/jwt');

const needTokenApi = [
    { method: "POST", path: "/api/v1/student" },
    { method: "PUT", path: "/api/v1/student/:id" },
    { method: "GET", path: "/api/v1/student" },
];

// 没有令牌
const hasNoToken = (res) => {
    res.status(401).send(getErr("你没有权限", 401));
}

module.exports = function (req, res, next) {
    // 校验白名单
    const apis = needTokenApi.filter(item => {
        const reg = pathToRegexp(item.path);
        return item.method == req.method && reg.regexp.test(req.path);
    })

    if (!apis.length) {
        next();
        return;
    }

    //验证token
    const decoded = jwtUtils.verify(req);
    console.log("校验jwt==>", req.headers.authorization, decoded);
    if (!decoded) {
        hasNoToken(res);
        return;
    }

    req.user = decoded.userId;
    next();
}
```

## 处理前端history模式
前端配置history模式，需要处理后端返回的index.html文件。

前端一般是SPA单页应用，当前端打包完之后，通常会将前端的资源放置到`public`目录下。

例如有一个地址为：
```sh
https://localhost/base/userinfo
```
当浏览器刷新的时候，就会出现找不到资源。原因是前端是单页应用，刷新的时候会请求`/base/userinfo`，而在服务器上并没有userinfo这个静态页。

所以需要服务器处理，返回index.html文件。

```sh
npm i connect-history-api-fallback -S
```

routes/init.js

```js
const express = require("express");
const app = express();

const history = require("connect-history-api-fallback");
app.use(history());//匹配不到资源的时候就会返回Index.html
```
